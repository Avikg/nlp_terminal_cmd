🔒 SAFETY FEATURES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

COMMANDS THAT DON'T REQUIRE CONFIRMATION:
  ✅ All read-only operations (listing, viewing, showing)
  ✅ Network diagnostics (ping, traceroute, nslookup)
  ✅ System information queries (CPU, memory, disk, uptime)
  ✅ File searches and directory listings
  ✅ Process listing (not killing)
  ✅ IP/MAC address queries
  ✅ Port checking

COMMANDS THAT REQUIRE CONFIRMATION:
  ⚠️ Deleting files or directories
  ⚠️ Modifying system files
  ⚠️ Killing processes
  ⚠️ Formatting drives
  ⚠️ Operations with sudo/admin privileges
  ⚠️ Network operations that send/modify data

💡 USAGE TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

BASIC USAGE:
  python -m nlpcmd_ai.cli "your natural language command"

INTERACTIVE MODE:
  python -m nlpcmd_ai.cli -i
  > what is my ip
  > show disk usage
  > exit

AUTO-CONFIRM (Skip Prompts):
  python -m nlpcmd_ai.cli --yes "your command"

DRY RUN (Preview Without Executing):
  python -m nlpcmd_ai.cli --dry-run "your command"

CREATE SHORTCUT (Windows):
  Create nlpai.bat:
  @echo off
  python -m nlpcmd_ai.cli %*
  
  Then use: nlpai "what is my ip"

CREATE ALIAS (Linux/macOS):
  Add to ~/.bashrc or ~/.zshrc:
  alias nlpai='python -m nlpcmd_ai.cli'
  
  Then use: nlpai "what is my ip"

# AI Provider (required)
NLP_PROVIDER=ollama  # or "openai" or "anthropic"

# Provider-specific settings
OLLAMA_MODEL=llama3.2
OPENAI_API_KEY=your-key-here
ANTHROPIC_API_KEY=your-key-here

# Safety settings
REQUIRE_CONFIRMATION=false  # Set to true for all confirmations

# Logging
LOG_COMMANDS=true
LOG_FILE=~/.nlpcmd_ai/history.log
